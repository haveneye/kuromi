// ☁️ 프로필 목록 로드 (로컬 + 클라우드 파이어스토어 동기화) window.loadProfiles = async function() { const savedProfiles = localStorage.getItem('sanrio_app_profiles_index'); window.profileList = savedProfiles ? JSON.parse(savedProfiles) : []; // 파이어스토어 클라우드에 저장된 프로필 목록도 함께 불러와 동기화 if (db && auth && auth.currentUser) { try { const configRef = doc(db, 'artifacts', appId, 'public', 'data', 'config', 'profiles'); const snap = await getDoc(configRef); if (snap.exists() && snap.data().list) { const cloudList = snap.data().list; // 로컬과 클라우드 목록 합치기 (중복 제거) window.profileList = Array.from(new Set([...window.profileList, ...cloudList])); localStorage.setItem('sanrio_app_profiles_index', JSON.stringify(window.profileList)); } } catch(e) { console.error("클라우드 프로필 목록 로드 실패:", e); } } }; // ☁️ 프로필 목록 저장 (로컬 + 클라우드 파이어스토어 동기화) window.saveProfilesIndex = async function() { localStorage.setItem('sanrio_app_profiles_index', JSON.stringify(window.profileList)); if (db && auth && auth.currentUser) { try { const configRef = doc(db, 'artifacts', appId, 'public', 'data', 'config', 'profiles'); await setDoc(configRef, { list: window.profileList }, { merge: true }); } catch(e) { console.error("클라우드 프로필 목록 저장 실패:", e); } } }; window.promptPinLogin = function(name) { window.targetLoginProfile = name; document.getElementById('pin-target-name').innerText = name; document.getElementById('pin-input').value = ''; document.getElementById('pin-login-modal').classList.remove('hidden'); }; window.closePinLoginModal = function() { document.getElementById('pin-login-modal').classList.add('hidden'); }; window.verifyPinLogin = function() { const enteredPin = document.getElementById('pin-input').value.trim(); const storageKey = `sanrio_app_profile_${window.targetLoginProfile}`; const saved = localStorage.getItem(storageKey); if (saved) { try { const parsed = JSON.parse(saved); const correctPin = parsed.pin || '0000'; if (enteredPin === correctPin) { window.closePinLoginModal(); window.loadProfileData(window.targetLoginProfile); } else { showToast("❌ 비밀번호가 틀렸습니다!"); } } catch(e) { showToast("❌ 데이터를 불러올 수 없습니다."); } } else { // 만약 로컬에 데이터가 없어도 클라우드(파이어스토어)에 프로필이 있다면 접속 허용 window.closePinLoginModal(); window.loadProfileData(window.targetLoginProfile); } };
